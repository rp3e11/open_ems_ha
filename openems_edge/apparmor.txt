#include <tunables/global>

profile openems_edge flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Allow all capabilities. The entrypoint needs many (chown, setuid,
  # sys_chroot, etc.) and the container is otherwise isolated by HA
  # Supervisor's container boundary. Following the same pattern as
  # most official HA add-ons (e.g. mosquitto).
  capability,

  # Networking required for Modbus-TCP, WebSocket, and Felix console.
  network,

  # JVM uses signals between threads.
  signal,

  # Allow ptrace within the container (JVM diagnostics, jstack, etc.).
  ptrace,

  # Filesystem: allow reading anywhere (needed for /etc/nsswitch.conf,
  # /etc/passwd, locale data, /lib, /usr/lib, JVM internals) and
  # writing where the entrypoint and OpenEMS need to.
  / r,
  /** r,
  /** mrwlk,
  /** ix,

  # Explicit denies for things nothing in this add-on should touch.
  # (/etc/shadow is intentionally NOT denied — useradd needs to write
  # to it on first start when creating the unprivileged openems user.)
  deny /etc/sudoers* w,
  deny /boot/** w,
}
